## Step 4: Implement AuthService (60-90 minutes)

### Create/Update `lib/services/auth_service.dart`:

```dart
import 'package:firebase_auth/firebase_auth.dart';
import 'package:flutter/material.dart';

class AuthService {
  final FirebaseAuth _auth = FirebaseAuth.instance;

  // Get current user
  User? get currentUser => _auth.currentUser;
  
  // Auth state stream
  Stream<User?> get authStateChanges => _auth.authStateChanges();
  
  // Get user ID
  String? get userId => _auth.currentUser?.uid;

  /// Sign up with email and password
  Future<UserCredential?> signUp({
    required String email,
    required String password,
    required String name,
  }) async {
    try {
      final userCredential = await _auth.createUserWithEmailAndPassword(
        email: email.trim(),
        password: password,
      );
      
      // Update display name
      await userCredential.user?.updateDisplayName(name);
      
      debugPrint('✅ User signed up: ${userCredential.user?.uid}');
      return userCredential;
      
    } on FirebaseAuthException catch (e) {
      debugPrint('❌ Sign up error: ${e.code}');
      throw _handleAuthException(e);
    }
  }

  /// Sign in with email and password
  Future<UserCredential?> signIn({
    required String email,
    required String password,
  }) async {
    try {
      final userCredential = await _auth.signInWithEmailAndPassword(
        email: email.trim(),
        password: password,
      );
      
      debugPrint('✅ User signed in: ${userCredential.user?.uid}');
      return userCredential;
      
    } on FirebaseAuthException catch (e) {
      debugPrint('❌ Sign in error: ${e.code}');
      throw _handleAuthException(e);
    }
  }

  /// Sign out
  Future<void> signOut() async {
    try {
      await _auth.signOut();
      debugPrint('✅ User signed out');
    } catch (e) {
      debugPrint('❌ Sign out error: $e');
      rethrow;
    }
  }

  /// Reset password
  Future<void> resetPassword({required String email}) async {
    try {
      await _auth.sendPasswordResetEmail(email: email.trim());
      debugPrint('✅ Password reset email sent to $email');
    } on FirebaseAuthException catch (e) {
      debugPrint('❌ Password reset error: ${e.code}');
      throw _handleAuthException(e);
    }
  }

  /// Check if user is authenticated
  Future<bool> isAuthenticated() async {
    return _auth.currentUser != null;
  }

  /// Handle Firebase Auth exceptions
  String _handleAuthException(FirebaseAuthException e) {
    switch (e.code) {
      case 'weak-password':
        return 'Password is too weak. Use at least 6 characters.';
      case 'email-already-in-use':
        return 'An account already exists with this email.';
      case 'user-not-found':
        return 'No account found with this email.';
      case 'wrong-password':
        return 'Incorrect password. Please try again.';
      case 'invalid-email':
        return 'Invalid email address format.';
      case 'user-disabled':
        return 'This account has been disabled.';
      case 'too-many-requests':
        return 'Too many attempts. Please try again later.';
      default:
        return 'Authentication error: ${e.message}';
    }
  }
}
```

```dart
import 'package:cloud_firestore/cloud_firestore.dart';
import '../models/user.dart';

class UserService {
  final FirebaseFirestore _db = FirebaseFirestore.instance;

  /// Create user profile
  Future<void> createUserProfile({
    required String userId,
    required String email,
    required String name,
  }) async {
    await _db.collection('users').doc(userId).set({
      'email': email,
      'name': name,
      'avatarUrl': '',
      'createdAt': FieldValue.serverTimestamp(),
      'plaidLinked': false,
      'plaidBankName': '',
      'friendIds': [],
      'pactIds': [],
    });
  }

  /// Get user profile
  Future<User?> getUserProfile(String userId) async {
    final doc = await _db.collection('users').doc(userId).get();
    if (doc.exists) {
      return User.fromFirestore(doc);
    }
    return null;
  }

  /// Update Plaid link status
  Future<void> updatePlaidStatus({
    required String userId,
    required bool linked,
    String bankName = '',
  }) async {
    await _db.collection('users').doc(userId).update({
      'plaidLinked': linked,
      'plaidBankName': bankName,
    });
  }

  /// Search users by email
  Future<List<User>> searchByEmail(String email) async {
    final query = await _db
        .collection('users')
        .where('email', isGreaterThanOrEqualTo: email.toLowerCase())
        .where('email', isLessThanOrEqualTo: '${email.toLowerCase()}\uf8ff')
        .limit(10)
        .get();

    return query.docs.map((doc) => User.fromFirestore(doc)).toList();
  }

  /// Send friend request
  Future<void> sendFriendRequest({
    required String fromUserId,
    required String toUserId,
  }) async {
    await _db.collection('friendRequests').add({
      'fromUserId': fromUserId,
      'toUserId': toUserId,
      'status': 'pending',
      'createdAt': FieldValue.serverTimestamp(),
    });
  }

  /// Accept friend request
  Future<void> acceptFriendRequest({
    required String userId,
    required String friendId,
  }) async {
    // Add to both users' friend lists
    await _db.collection('users').doc(userId).update({
      'friendIds': FieldValue.arrayUnion([friendId]),
    });
    await _db.collection('users').doc(friendId).update({
      'friendIds': FieldValue.arrayUnion([userId]),
    });
  }

  /// Get friends list
  Future<List<User>> getFriends(String userId) async {
    final userDoc = await _db.collection('users').doc(userId).get();
    final friendIds = List<String>.from(userDoc.data()?['friendIds'] ?? []);

    if (friendIds.isEmpty) return [];

    final friendDocs = await Future.wait(
      friendIds.map((id) => _db.collection('users').doc(id).get()),
    );

    return friendDocs
        .where((doc) => doc.exists)
        .map((doc) => User.fromFirestore(doc))
        .toList();
  }
}
```

## Step 3: Create PactService (90 minutes)

### Update `lib/services/pact_service.dart`:

```dart
import 'package:cloud_firestore/cloud_firestore.dart';
import '../models/pact.dart';
import '../models/contribution.dart';

class PactService {
  final FirebaseFirestore _db = FirebaseFirestore.instance;

  /// Create new pact
  Future<void> createPact(Pact pact) async {
    await _db.collection('pacts').doc(pact.id).set(pact.toMap());
    
    // Add pact to creator's pact list
    await _db.collection('users').doc(pact.createdBy).update({
      'pactIds': FieldValue.arrayUnion([pact.id]),
    });
  }

  /// Get user's pacts
  Future<List<Pact>> getUserPacts({required String userId}) async {
    final query = await _db
        .collection('pacts')
        .where('memberIds', arrayContains: userId)
        .where('isActive', isEqualTo: true)
        .orderBy('createdAt', descending: true)
        .get();

    return query.docs.map((doc) => Pact.fromFirestore(doc)).toList();
  }

  /// Get pact by ID
  Future<Pact?> getPactById(String pactId) async {
    final doc = await _db.collection('pacts').doc(pactId).get();
    if (doc.exists) {
      return Pact.fromFirestore(doc);
    }
    return null;
  }

  /// Add member to pact
  Future<void> addMember({
    required String pactId,
    required String userId,
  }) async {
    await _db.collection('pacts').doc(pactId).update({
      'memberIds': FieldValue.arrayUnion([userId]),
    });
    
    await _db.collection('users').doc(userId).update({
      'pactIds': FieldValue.arrayUnion([pactId]),
    });
  }

  /// Add contribution
  Future<void> addContribution(Contribution contribution) async {
    final batch = _db.batch();

    // Add contribution document
    final contributionRef = _db.collection('contributions').doc();
    batch.set(contributionRef, contribution.toMap());

    // Update pact current amount
    final pactRef = _db.collection('pacts').doc(contribution.pactId);
    batch.update(pactRef, {
      'currentAmount': FieldValue.increment(contribution.amount),
    });

    await batch.commit();
  }

  /// Get contributions for a pact
  Stream<List<Contribution>> getContributionsStream(String pactId) {
    return _db
        .collection('contributions')
        .where('pactId', isEqualTo: pactId)
        .orderBy('date', descending: true)
        .snapshots()
        .map((snapshot) => snapshot.docs
            .map((doc) => Contribution.fromFirestore(doc))
            .toList());
  }

  /// Get leaderboard (top contributors)
  Future<List<Map<String, dynamic>>> getLeaderboard({
    String? pactId,
    int limit = 10,
  }) async {
    Query query = _db.collection('contributions');
    
    if (pactId != null) {
      query = query.where('pactId', isEqualTo: pactId);
    }

    final snapshot = await query.get();
    
    // Group by user and sum amounts
    final Map<String, double> userTotals = {};
    final Map<String, String> userNames = {};
    
    for (var doc in snapshot.docs) {
      final contribution = Contribution.fromFirestore(doc);
      userTotals[contribution.userId] = 
          (userTotals[contribution.userId] ?? 0) + contribution.amount;
      // You'll need to fetch user names separately
    }

    // Sort and return top contributors
    final leaderboard = userTotals.entries
        .map((entry) => {
              'userId': entry.key,
              'totalAmount': entry.value,
            })
        .toList()
      ..sort((a, b) => (b['totalAmount'] as double)
          .compareTo(a['totalAmount'] as double));

    return leaderboard.take(limit).toList();
  }
}
```

```dart
    userId: data['userId'] ?? '',
```

### Add Security Rules (Production):
```
rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    // Users
    match /users/{userId} {
      allow read: if request.auth != null;
      allow write: if request.auth.uid == userId;
    }
    
    // Pacts
    match /pacts/{pactId} {
      allow read: if request.auth.uid in resource.data.memberIds;
      allow create: if request.auth != null;
      allow update: if request.auth.uid in resource.data.memberIds;
    }
    
    // Contributions
    match /contributions/{contributionId} {
      allow read: if request.auth != null;
      allow create: if request.auth.uid == request.resource.data.userId;
    }
    
    // Friend Requests
    match /friendRequests/{requestId} {
      allow read, write: if request.auth != null;
    }
  }
}
```
